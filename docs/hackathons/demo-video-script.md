```sh
./.venv/bin/python -m pytest tests/test_hackathon_verifiability.py tests/test_submit_mantle_decisions.py
```

```sh
./.venv/bin/python scripts/submit_mantle_decisions.py \
  --commitments data/mantle_commitments.jsonl \
  --contract "$AGENT_REGISTRY_ADDRESS" \
  --agent-id "$AGENT_ID" \
  --limit 1
```
